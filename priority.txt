minheap::minheap()
{
   heapsize=0;
   capacity=10;
   heap=new CAnime[capacity];
   for(int i=0;i<capacity;i++)
   {
      heap[i]=CAnime();
   }
}
minheap::minheap(int h)
{
   capacity=pow(2,h+1)-1;
   heap=new CAnime[capacity+1];
   heapsize=0;
}
minheap::~minheap()
{
   heapsize=0;
   capacity=0;
   delete[]heap;
   heap=NULL;
}
int minheap::isFull()
{
   if(heapsize==capacity)
      return 1;
   else
      return 0;
}
int minheap::isEmpty()
{
   if(heapsize<1)
      return 1;
   else
      return 0;
}
void minheap::percolateUp(int i)
{
   int parent=i/2;
   CAnime temp=heap[i];
   while(i>1 && heap[parent].rate>temp.rate)
   {
      heap[i]=heap[parent];
      i=parent;
      parent=i/2;
   }
   heap[i]=temp;
}
int minheap::insert(CAnime a)
{
   if(a.rate<0)
   {
      return -2;
   }
   else if(isFull())
   {
      cout<<"Heap is full"<<endl;
      return -1;
   }
   heap[++heapsize]=a;
   percolateUp(heapsize);
   return 1;
}
void minheap::display()
{
    if(heapsize!=0)
    {
      cout<<"Elements:"<<endl;
      for(int i=1;i<=heapsize;i++)
      {
         heap[i].display();
      }
   }
   else
   {
      cout<<"There is no element in the heap"<<endl;
   }
}
void minheap::percolateDown(int i)
{
    int child;
    CAnime temp=heap[i];
    while(2*i<=heapsize)
    {
       child=2*i;
       if(child!=heapsize && heap[child+1].rate<heap[child].rate)
       {
          child++;
       }
       if(heap[child].rate<temp.rate)
       {
          heap[i]=heap[child];
          i=child;
       }
       else
          break;
    }
    heap[i]=temp;
}
int minheap::deleteMin()
{
    if(isEmpty())
    {
        return 0;
    }
    heap[1]=heap[heapsize--];
    percolateDown(1);
    return 1;
}
int minheap::search(int rate)
{
    for(int i=1;i<=heapsize;i++)
    {
        if(heap[i].rate==rate)
        {
            return i;
        }
    }
    return -1;
}
int minheap::increaseByKey(int ele,int inc)
{
   int index=search(ele);
   if(index==-1)
   {
      return -1;
   }
   else if(inc<0)
   {
      return -2;
   }

   else
   {
      heap[index].rate=ele+inc;
      percolateDown(index);
      return 1;
   }
}
int minheap::increaseWithKey(int ele,int key)
{
   int index=search(ele);
   if(index==-1)
   {
      return -1;
   }
   else if(key<0)
   {
      return -2;
   }
   else
   {
      if(key<=heap[index].rate)
      {
         return -3;
      }
      else
      {
         heap[index].rate=key;
         percolateDown(index);
         return 1;
      }
   }
}
int minheap::decreaseByKey(int ele,int dec)
{
   int index=search(ele);
   if(index==-1)
   {
      return -1;
   }
   else if(dec<0)
   {
      return -2;
   }
   else
   {
      heap[index].rate=ele-dec;
      percolateUp(index);
      return 1;
   }
}

int minheap::decreaseWithKey(int ele,int key)
{
   int index=search(ele);
   if(index==-1)
   {
      return -1;
   }
   else if(key<0)
   {
      return -2;
   }
   else
   {
      if(key>=heap[index].rate)
      {
         return -2;
      }
      else
      {
         heap[index].rate=key;
         percolateUp(index);
         return 1;
      }
   }
}
int minheap::buildheap(CAnime arr[],int n)
{
   if(n>capacity)
   {
      return -1;
   }
   heapsize=n;
   for(int i=1;i<=n;i++)
   {
      heap[i]=arr[i];
   }
   for(int i=heapsize/2;i>=1;i--)
   {
      percolateDown(i);
   }
   return 1;
}
