{
   public:
      CAList();

      int isFull();
      int isEmpty();
      int displayList();
      int insertAtFirst(int);
      int insertAtLast(int);
      int insertByPosition(int,int);
      int deleteAtFirst();
      int deleteAtLast();
      int deleteByPosition(int);
      int insertAtMiddle(int);
      int makeListEmpty();
      int searchEltByVal(int);
      int getElement(int);
      int Update(int,int);
   private:
      int *arr;
      int capacity;
      int size;
};

#include "head.h"
CAList::CAList()
{
   size=0;
   capacity=10;
   arr=new int[capacity];
   for(int i=0;i<capacity;i++)
   {
      arr[i]=-1;
   }
}
CAList::CAList(int *a,int c,int s)
{
   size=s;
   capacity=c;
   arr=new int[capacity];
   for(int i=0;i<s;i++)
   {
      arr[i]=a[i];
   }
   for(int i=s;i>capacity;i++)
   {
      arr[i]=-1;
   }
}

int CAList::isFull()
{
   if(size==capacity)
      return 1;
   else
      return 0;
}

int CAList::isEmpty()
{
   if(size==0)
      return 1;
   else
      return 0;
}

int CAList::displayList()
{
   if(isEmpty())
      return -1;
   else
   for(int i=0;i<size;i++)
   {
      cout<<arr[i]<<"\t";
   }
   cout<<endl;
   return 1;
}

int CAList::insertAtFirst(int ele)
{
   if (isFull())
      return -1;
   else if(isEmpty())
   {
      arr[0]=ele;
      size++;
      return 1;
   }
   else
   {
      for(int i=size;i>0;i--)
      {
         arr[i]=arr[i-1];
      }
      arr[0]=ele;
      size++;
      return 2;
   }
}

int CAList::insertAtLast(int ele)
{
   if (isFull())
      return -1;
   else
   {
      arr[size]=ele;
      size++;
      return 1;
   }
}

int CAList::deleteAtFirst()
{
   if (isEmpty())
      return -1;
   else
   {
      for(int i=0;i<size-1;i++)
      {
         arr[i]=arr[i+1];
      }
      arr[size-1]=-1;
      size--;
      return 1;
   }
}

int CAList::deleteAtLast()
{
   if (isEmpty())
      return -1;
   else
   {
      arr[size-1]=-1;
      size--;
      return 1;
   }
}

int CAList::insertByPosition(int ele,int pos)
{
   if(pos<0 || pos>size+1)
      return -1;
   else if(isFull())
      return -2;
   else if(isEmpty()&&pos==1)
   {
      arr[0]=ele;
      size++;
      return 1;
   }
   else if(isEmpty()&&pos!=1)
      return -3;
   else
   {
      for(int i=size;i>pos-1;i--)
      {
         arr[i]=arr[i-1];
      }
      arr[pos-1]=ele;
      size++;
      return 2;
   }
}

int CAList::deleteByPosition(int pos)
{
   if(pos<=0||pos>size+1)
      return -1;
   else if(isEmpty())
      return -2;
   else
   {
      for(int i=pos-1;i<size-1;i++)
      {
         arr[i]=arr[i+1];
      }
      size--;
      return 1;
   }
}

int CAList::makeListEmpty()
{
   if(isEmpty())
      return -1;
   else
   {
      while(!isEmpty())
         deleteAtFirst();
   }
   return 1;
}

int CAList::insertAtMiddle(int ele)
{
   if (isFull())
      return -1;
   else if(isEmpty())
   {
      arr[0]=ele;
      size++;
      return 1;
   }
   else
   {
      for(int i=size;i>size/2;i--)
      {
         arr[i]=arr[i-1];
      }
      arr[size/2]=ele;
      size++;
      return 2;
   }
}

int CAList::searchEltByVal(int elt)
{
      if(isEmpty())
      {
         return -1;
      }
      else
      {
         for(int i=0;i<size;i++)
         {
            if (arr[i]==elt)
            {
               return i;
            }

         }
         return 0;
      }
}

int CAList::Update(int elt,int elt1)
{
   if(isEmpty())
      return -1;
   else
   {
      for(int i=0;i<size;i++)
      {
         if(arr[i]==elt)
         {
            arr[i]=elt1;
            return 1;
         }
      }
      return -2;
   }
}
